First and Last Occurrences
Difficulty: MediumAccuracy: 37.36%Submissions: 302K+Points: 4Average Time: 15m
Given a sorted array arr with possibly some duplicates, the task is to find the first and last occurrences of an element x in the given array.
Note: If the number x is not found in the array then return both the indices as -1.

Examples:

Input: arr[] = [1, 3, 5, 5, 5, 5, 67, 123, 125], x = 5
Output: [2, 5]
Explanation: First occurrence of 5 is at index 2 and last occurrence of 5 is at index 5
Input: arr[] = [1, 3, 5, 5, 5, 5, 7, 123, 125], x = 7
Output: [6, 6]
Explanation: First and last occurrence of 7 is at index 6
Input: arr[] = [1, 2, 3], x = 4
Output: [-1, -1]
Explanation: No occurrence of 4 in the array, so, output is [-1, -1]
Constraints:
1 ≤ arr.size() ≤ 106
1 ≤ arr[i], x ≤ 109

class GFG {
    ArrayList<Integer> find(int arr[], int x) {
        // code here
        
        int low = 0;
        int high = arr.length-1;
        int min = -1;
        int max = -1;
        
        min = firstOccurance(arr,x,high,low);
        
        if(min==-1){
            return new ArrayList<>(List.of(min,max));
        }
        
        max = lastOccurance(arr,x,high,min);

        
        return new ArrayList<>(List.of(min,max));
       
    }
    
    int firstOccurance(int[] arr,int x,int high,int low) {
        int ans = -1;
        while(low<=high){
            int mid = low + (high-low)/2;
            
            if(arr[mid]<x){
                low = mid+1;
            }else if(arr[mid]>x){
                high = mid-1;
            }else{
                high = mid-1;
                ans = mid;
            }
        }
        return ans;
    }
    
    int lastOccurance(int[] arr,int x,int high,int low) {
        int ans = -1;
        while(low<=high){
            int mid = low + (high-low)/2;
            
            if(arr[mid]>x){
                high = mid-1;
            }else{
                low = mid+1;
                ans = mid;
            }
        }
        return ans;
    }
}
